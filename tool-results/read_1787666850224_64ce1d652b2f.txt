    33→
    34→/** Polar → Cartesian: 0° = North (top), clockwise */
    35→function pol(cx: number, cy: number, r: number, deg: number) {
    36→  const rad = (deg * Math.PI) / 180;
    37→  return { x: cx + r * Math.sin(rad), y: cy - r * Math.cos(rad) };
    38→}
    39→
    40→/** Text rotation so label is never upside-down */
    41→function textRot(deg: number) {
    42→  return deg > 90 && deg < 270 ? deg - 180 : deg;
    43→}
